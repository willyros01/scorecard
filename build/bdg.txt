#!/usr/bin/env bash
# The Scorecard — SPENDING ALERT (prepared by Claude, Oct 1, 2026)
#
# Run:  bash bdg.txt
#
# Sets ONE budget of 1 dollar on your Google billing account. It covers every
# project on that account (Scorecard, FIDUNIO and the third one). Google emails
# you when the month's spending reaches 50%, 90% and 100% of it.
# It only warns. It switches nothing off and changes no project.
# Safe to run again: if the budget is already there, it says so and stops.

# ---------------------------------------------------------------------------
# THE GOOGLE PROJECT — defined first, before any command runs.
# ---------------------------------------------------------------------------
PROJECT_ID="scorecard-f41b8"
unset CLOUDSDK_CORE_PROJECT GOOGLE_CLOUD_PROJECT

set -Eeuo pipefail

echo "Switching Cloud Shell to project ${PROJECT_ID} ..."
gcloud config set project "${PROJECT_ID}" --quiet >/dev/null 2>&1 || { echo "ERROR: could not switch to project ${PROJECT_ID}. Nothing changed." >&2; exit 1; }
CURRENT_PROJECT="$(gcloud config get-value project 2>/dev/null)"
[[ "${CURRENT_PROJECT}" == "${PROJECT_ID}" ]] || { echo "ERROR: Cloud Shell is on '${CURRENT_PROJECT}', not ${PROJECT_ID}. Nothing changed." >&2; exit 1; }
echo "OK: Cloud Shell is now on project ${CURRENT_PROJECT}."
export CLOUDSDK_CORE_PROJECT="${PROJECT_ID}"
export GOOGLE_CLOUD_PROJECT="${PROJECT_ID}"

NAME="Spending alert - 1 dollar"
AMOUNT="1.00"

die(){ echo "ERROR: $*" >&2; exit 1; }
command -v gcloud >/dev/null || die "gcloud is unavailable. Run this in Google Cloud Shell."
ACCOUNT="$(gcloud auth list --filter=status:ACTIVE --format='value(account)' | head -n1)"
[[ -n "${ACCOUNT}" ]] || die "No active Google account. Sign in to Cloud Shell first."

# The budget tool talks to Google through this project; switching it on is free.
echo "Making sure the budget tool is switched on ..."
gcloud services enable billingbudgets.googleapis.com --project="${PROJECT_ID}" --quiet >/dev/null \
  || die "Could not switch on the budget tool. Nothing changed. Send a screenshot to Claude."

# The billing account: there must be exactly one open account.
mapfile -t ACCOUNTS < <(gcloud billing accounts list --filter=open=true --format='value(name)' | sed 's#^billingAccounts/##')
[[ "${#ACCOUNTS[@]}" -ge 1 && -n "${ACCOUNTS[0]:-}" ]] || die "No open billing account was found for ${ACCOUNT}. Nothing changed."
[[ "${#ACCOUNTS[@]}" -eq 1 ]] || die "More than one billing account was found (${ACCOUNTS[*]}). Nothing changed. Send a screenshot to Claude."
BILLING="${ACCOUNTS[0]}"
CURRENCY="$(gcloud billing accounts describe "${BILLING}" --format='value(currencyCode)')"
[[ -n "${CURRENCY}" ]] || die "Could not read the account's currency. Nothing changed."
echo
echo "  Google account  : ${ACCOUNT}"
echo "  Billing account : ${BILLING} (${CURRENCY})"
echo

if gcloud billing budgets list --billing-account="${BILLING}" --format='value(displayName)' | grep -Fxq "${NAME}"; then
  echo "OK: the budget \"${NAME}\" is already there. Nothing else to do."
  exit 0
fi

echo "Creating the budget: ${AMOUNT} ${CURRENCY} a month, emails at 50%, 90% and 100% ..."
gcloud billing budgets create --billing-account="${BILLING}" --display-name="${NAME}" \
  --budget-amount="${AMOUNT}${CURRENCY}" \
  --threshold-rule=percent=0.5 --threshold-rule=percent=0.9 --threshold-rule=percent=1.0 >/dev/null \
  || die "Google refused the budget. Nothing was created. Send a screenshot to Claude."

gcloud billing budgets list --billing-account="${BILLING}" --format='value(displayName)' | grep -Fxq "${NAME}" \
  || die "The budget was not found after creating it. Send a screenshot to Claude."
echo "OK: the budget is in place."
echo
echo "DONE. Google will email ${ACCOUNT} if a month's spending reaches 50, 90 or 100 cents."
