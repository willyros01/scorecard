#!/usr/bin/env bash
# The Scorecard — VERSION 2.0 GO-LIVE (prepared by Claude, Sep 30, 2026)
#
# Apply:    bash go2.txt
# Verify:   bash go2.txt verify
# Status:   bash go2.txt status      (how far go-live got)
# Rollback: bash go2.txt rollback
#
# What it does, in this order (it shows the plan and waits for you to type
# yes before changing anything):
#   1. checks the published rules are exactly the ones rf1.txt published;
#   2. prepares the data for Version 2.0 (build/golive.mjs): you as the only
#      group creator, the public group, admin codes moved to where only the
#      owner can read them, each golfer's groups, each group's directory of
#      names and indexes, each membership's golfer, each game's players;
#   3. publishes the Version 2.0 rules;
#   4. checks everything is in place;
#   5. changes the subject of Firebase's password email (optional; if Google
#      refuses it, the default wording simply stays).
# From step 3 the OLD web app (2.21.9) stops working for members. Tell Claude
# straight away to switch the web to Version 2.0.
# If any step fails, the script puts everything back by itself (the old rules
# and the admin codes) and says so; nothing is lost and it can be run again.
# The database keeps a marker of how far it got (migrations/v2).
# Rollback puts back the old rules, the admin codes and the email subject.
#
# Run it in Google Cloud Shell, signed in with the Google account that owns
# the Firebase project scorecard-f41b8.

# ---------------------------------------------------------------------------
# THE GOOGLE PROJECT — defined first, before any command runs.
# ---------------------------------------------------------------------------
PROJECT_ID="scorecard-f41b8"
unset CLOUDSDK_CORE_PROJECT GOOGLE_CLOUD_PROJECT

set -Eeuo pipefail

echo "Switching Cloud Shell to project ${PROJECT_ID} ..."
gcloud config set project "${PROJECT_ID}" --quiet >/dev/null 2>&1 || { echo "ERROR: could not switch to project ${PROJECT_ID}. Nothing changed." >&2; exit 1; }
CURRENT_PROJECT="$(gcloud config get-value project 2>/dev/null)"
[[ "${CURRENT_PROJECT}" == "${PROJECT_ID}" ]] || { echo "ERROR: Cloud Shell is on '${CURRENT_PROJECT}', not ${PROJECT_ID}. Nothing changed." >&2; exit 1; }
echo "OK: Cloud Shell is now on project ${CURRENT_PROJECT}."
export CLOUDSDK_CORE_PROJECT="${PROJECT_ID}"
export GOOGLE_CLOUD_PROJECT="${PROJECT_ID}"

MODE="${1:-apply}"
OWNER_EMAIL="willyros01@gmail.com"
COMMIT="10e89c2dbbce20c22d036fae94a4503410ae6836"                                        # v2 branch, tested
LIVE_COMMIT="58948239af100eac8b9a80df95af0ceff4959931"     # ios branch: the rules rf1.txt published
RAW="https://raw.githubusercontent.com/willyros01/scorecard"
WEB_KEY="AIzaSyA5qWtnz3QHco5YSAlHb7TXmCNg1kUdVhA"
STATE="$(pwd)/scorecard-go2-backup"
WORK="$(pwd)/scorecard-go2-tools"
RULES_API="https://firebaserules.googleapis.com/v1"
AUTH_CONFIG="https://identitytoolkit.googleapis.com/admin/v2/projects/${PROJECT_ID}/config"
EMAIL_SUBJECT="Choose your password for The Scorecard"
EMAIL_SENDER="The Scorecard"

die(){ echo "ERROR: $*" >&2; exit 1; }
for c in gcloud curl jq python3 node diff; do command -v "$c" >/dev/null || die "$c is unavailable. Run this in Google Cloud Shell."; done
[[ "$(node -p 'process.versions.node.split(".")[0]')" -ge 18 ]] || die "Node 18 or newer is needed."

ACCOUNT="$(gcloud auth list --filter=status:ACTIVE --format='value(account)' | head -n1)"
[[ -n "${ACCOUNT}" ]] || die "No active Google account. Sign in to Cloud Shell first."
FOUND_ID="$(gcloud projects describe "${PROJECT_ID}" --project="${PROJECT_ID}" --format='value(projectId)' 2>/dev/null || true)"
[[ "${FOUND_ID}" == "${PROJECT_ID}" ]] || die "Project ${PROJECT_ID} was not found, or ${ACCOUNT} cannot see it. Nothing changed."
APP_CONFIG="$(curl --fail --silent --show-error "${RAW}/${COMMIT}/firebase-config.js")" || die "Cannot read firebase-config.js from GitHub. Nothing changed."
grep -Fq "projectId: \"${PROJECT_ID}\"" <<<"${APP_CONFIG}" || die "firebase-config.js does not name ${PROJECT_ID}. Nothing changed."
grep -Fq "apiKey: \"${WEB_KEY}\"" <<<"${APP_CONFIG}" || die "firebase-config.js does not use the expected browser key. Nothing changed."
echo
echo "  Google account : ${ACCOUNT}"
echo "  Project        : ${PROJECT_ID}"
echo "  Owner          : ${OWNER_EMAIL}"
echo

token(){ gcloud auth print-access-token; }
HDR_PROJECT="x-goog-user-project: ${PROJECT_ID}"
get(){ curl --fail-with-body --silent --show-error -H "Authorization: Bearer $(token)" -H "${HDR_PROJECT}" "$1"; }
send(){ curl --fail-with-body --silent --show-error -X "$1" -H "Authorization: Bearer $(token)" -H "${HDR_PROJECT}" -H 'Content-Type: application/json' --data-binary "@$3" "$2"; }
release_ruleset(){ get "${RULES_API}/projects/${PROJECT_ID}/releases/cloud.firestore" | jq -er .rulesetName; }
save_rules(){ get "${RULES_API}/$1" | jq -er '.source.files[]|select(.name=="firestore.rules")|.content' > "$2"; [[ -s "$2" ]] || die "Empty Firestore rules."; }
same_rules(){ python3 - "$1" "$2" <<'PYC'
import sys
a=open(sys.argv[1]).read().rstrip("\n"); b=open(sys.argv[2]).read().rstrip("\n")
sys.exit(0 if a==b else 1)
PYC
}
publish_file(){
  local body="${STATE}/ruleset.json" patch="${STATE}/release.json" ruleset
  jq -n --rawfile rules "$1" '{source:{files:[{name:"firestore.rules",content:$rules}]}}' > "${body}" || return 1
  # Creating the ruleset compiles it; a rules error stops here, before anything is published.
  ruleset="$(send POST "${RULES_API}/projects/${PROJECT_ID}/rulesets" "${body}" | jq -er .name)" || return 1
  [[ -n "${ruleset}" ]] || return 1
  jq -n --arg name "projects/${PROJECT_ID}/releases/cloud.firestore" --arg ruleset "${ruleset}" '{release:{name:$name,rulesetName:$ruleset},updateMask:"rulesetName"}' > "${patch}" || return 1
  send PATCH "${RULES_API}/projects/${PROJECT_ID}/releases/cloud.firestore" "${patch}" >/dev/null || return 1
}
# The live rules equal this file? (no exit on failure)
live_is(){
  local current
  current="$(release_ruleset 2>/dev/null)" || return 1
  get "${RULES_API}/${current}" 2>/dev/null | jq -er '.source.files[]|select(.name=="firestore.rules")|.content' > "${STATE}/rules-live-now.txt" 2>/dev/null || return 1
  same_rules "$1" "${STATE}/rules-live-now.txt"
}
# Automatic recovery: the earlier rules back (tried 3 times), the admin codes
# back, the marker set, and a plain message. Always ends the script.
recover(){
  local what="$1" tries=0 back=no
  echo
  echo "Something failed during ${what}. Putting everything back automatically ..."
  while (( tries < 3 )); do
    tries=$((tries + 1))
    if live_is "${STATE}/rules-before.txt"; then back=yes; break; fi
    publish_file "${STATE}/rules-before.txt" || true
    sleep 3
  done
  rm -f "${STATE}/rules-published"
  data mark failed "${what}" >/dev/null 2>&1 || true
  data rollback || echo "Note: the admin codes could not be put back yet. Run: bash go2.txt rollback"
  if [[ "${back}" == yes ]]; then
    echo "DONE: the earlier rules are live again and the old web app works. Nothing was lost."
    echo "Send a screenshot to Claude. When it is fixed, bash go2.txt can simply be run again."
  else
    echo "WARNING: the earlier rules could not be confirmed. Run:  bash go2.txt rollback  and send a screenshot to Claude."
  fi
  exit 1
}
data(){ ACCESS_TOKEN="$(token)" OWNER_EMAIL="${OWNER_EMAIL}" PROJECT="${PROJECT_ID}" STATE_DIR="${STATE}" \
  MODEL="${WORK}/model.mjs" node "${WORK}/golive.mjs" "$@"; }

mkdir -p "${STATE}" "${WORK}"
curl --fail --silent --show-error "${RAW}/${COMMIT}/build/golive.mjs" -o "${WORK}/golive.mjs" || die "Cannot read golive.mjs from GitHub. Nothing changed."
curl --fail --silent --show-error "${RAW}/${COMMIT}/model.js" -o "${WORK}/model.mjs" || die "Cannot read model.js from GitHub. Nothing changed."
curl --fail --silent --show-error "${RAW}/${COMMIT}/firestore.rules" -o "${WORK}/rules-v2.txt" || die "Cannot read the Version 2.0 rules from GitHub. Nothing changed."
grep -Fq "PHASE D" "${WORK}/rules-v2.txt" || die "The downloaded rules are not the Version 2.0 rules. Nothing changed."

verify_all(){
  save_rules "$(release_ruleset)" "${STATE}/rules-live-now.txt"
  same_rules "${WORK}/rules-v2.txt" "${STATE}/rules-live-now.txt" || die "The published rules are not the Version 2.0 rules."
  echo "OK: the published rules are the Version 2.0 rules."
  data verify
}

if [[ "${MODE}" == apply ]]; then
  if [[ -e "${STATE}/rules-published" ]]; then
    echo "Version 2.0 was already published in an earlier run. Checking only."
    verify_all; echo; echo "DONE. Nothing else to do."; exit 0
  fi
  # 1. The published rules must be exactly the ones rf1.txt published.
  [[ -s "${STATE}/rules-before.txt" ]] || save_rules "$(release_ruleset)" "${STATE}/rules-before.txt"
  curl --fail --silent --show-error "${RAW}/${LIVE_COMMIT}/firestore.rules" -o "${WORK}/rules-live-expected.txt" || die "Cannot read the current rules from GitHub. Nothing changed."
  same_rules "${STATE}/rules-before.txt" "${WORK}/rules-live-expected.txt" || die "The published rules are not the ones rf1.txt published. Nothing changed. Send a screenshot to Claude."
  echo "OK: the published rules are the ones rf1.txt published (saved for rollback)."
  echo
  # 2. The plan, then your go-ahead.
  data plan
  echo
  echo "From here the old web app (2.21.9) stops working for members until the web is switched to Version 2.0."
  read -r -p "Type yes to go live: " ANSWER
  [[ "${ANSWER}" == "yes" ]] || die "Stopped. Nothing was changed."
  data apply || recover "the data step"
  # 3. The Version 2.0 rules.
  publish_file "${WORK}/rules-v2.txt" || recover "publishing the rules"
  touch "${STATE}/rules-published"
  data mark rules-published >/dev/null 2>&1 || true
  sleep 3
  # 4. Check: the rules (twice, as Google can take a moment), then the data.
  live_is "${WORK}/rules-v2.txt" || { sleep 10; live_is "${WORK}/rules-v2.txt"; } || recover "checking the published rules"
  echo "OK: the published rules are the Version 2.0 rules."
  data verify || recover "the final data check"
  data mark verified >/dev/null 2>&1 || true
  # 5. The email subject (optional).
  if [[ ! -s "${STATE}/auth-config-before.json" ]]; then get "${AUTH_CONFIG}" > "${STATE}/auth-config-before.json" || true; fi
  jq -n --arg s "${EMAIL_SUBJECT}" --arg n "${EMAIL_SENDER}" \
    '{notification:{sendEmail:{resetPasswordTemplate:{subject:$s,senderDisplayName:$n}}}}' > "${STATE}/email.json"
  if send PATCH "${AUTH_CONFIG}?updateMask=notification.sendEmail.resetPasswordTemplate.subject,notification.sendEmail.resetPasswordTemplate.senderDisplayName" "${STATE}/email.json" >/dev/null 2>&1; then
    echo "OK: the password email now has the subject \"${EMAIL_SUBJECT}\"."
  else
    echo "Note: Google did not accept the new email subject. The default wording stays. Nothing else is affected."
  fi
  echo
  echo "DONE. Version 2.0 is live. Now tell Claude to switch the web to Version 2.0."
elif [[ "${MODE}" == verify ]]; then
  verify_all
elif [[ "${MODE}" == status ]]; then
  data status
elif [[ "${MODE}" == rollback ]]; then
  [[ -s "${STATE}/rules-before.txt" ]] || die "No saved rules to go back to."
  publish_file "${STATE}/rules-before.txt"
  rm -f "${STATE}/rules-published"
  save_rules "$(release_ruleset)" "${STATE}/rules-live-now.txt"
  same_rules "${STATE}/rules-before.txt" "${STATE}/rules-live-now.txt" || die "Rollback of the rules did not take. Send a screenshot to Claude."
  echo "OK: the earlier rules are published again."
  data rollback
  if [[ -s "${STATE}/auth-config-before.json" ]]; then
    jq '{notification:{sendEmail:{resetPasswordTemplate:(.notification.sendEmail.resetPasswordTemplate // {})}}}' "${STATE}/auth-config-before.json" > "${STATE}/email-back.json"
    send PATCH "${AUTH_CONFIG}?updateMask=notification.sendEmail.resetPasswordTemplate.subject,notification.sendEmail.resetPasswordTemplate.senderDisplayName" "${STATE}/email-back.json" >/dev/null 2>&1 \
      && echo "OK: the email subject is back as it was." || echo "Note: the email subject could not be put back; it is harmless."
  fi
  echo "Now tell Claude to put the web back to 2.21.9."
else
  die "Unknown mode ${MODE}. Use: bash go2.txt   or   bash go2.txt verify   or   bash go2.txt status   or   bash go2.txt rollback"
fi
